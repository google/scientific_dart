import 'package:ndarray/ndarray.dart';

late final NDArray<Float64> f64;
late final NDArray<Float32> f32;
late final NDArray<Int64> i64;
late final NDArray<Int32> i32;
late final NDArray<Uint8> u8;
late final NDArray<Boolean> b;
late final NDArray<DTypeTag> dyn;

void isSubtype<Sub extends Super, Super>() {}

// Acceptance-criteria negative probes (R3b):
final t_neg_bitwiseAnd_f64 = bitwiseAnd(f64, f64); // error: argument_type_not_assignable
final t_neg_opand_f64 = f64 & f64; // error: undefined_operator
final t_neg_leftShift_b = leftShift(b, b); // error: argument_type_not_assignable
final t_neg_opshl_b = b << b; // error: undefined_operator
final t_neg_sin_dyn = sin(dyn); // error: argument_type_not_assignable

// Additional negative probes on bare DTypeTag and invalid operator receivers:
final t_neg_sum_dyn = sum(dyn); // error: argument_type_not_assignable
final t_neg_opand_dyn = dyn & dyn; // error: undefined_operator
final t_neg_opshl_f32 = f32 << f32; // error: undefined_operator

// Negative probes for mixed-dtype binary & multi-array operations (SelfOf):
final t_neg_add_f64_f32 = add(f64, f32); // error: could_not_infer
final t_neg_add_f32_f64 = add(f32, f64); // error: could_not_infer

final t_neg_add_out_f32 = add(f64, f64, out: f32); // error: could_not_infer
final t_neg_divide_i64_i32 = divide(i64, i32); // error: could_not_infer
final t_neg_atan2_i64_i32 = atan2(i64, i32); // error: could_not_infer

// LUB(Float32, Int32) is `DoublePrecisionOf<Float64>` (their only shared
// depth-2 interface), which satisfied the pre-SelfOf `T extends
// DoublePrecisionOf<R>` bound of floatPower/cov; SelfOf closes that hole.
final t_neg_floatPower_f32_i32 = floatPower(f32, i32); // error: could_not_infer
final t_neg_cov_f32_i32 = cov(f32, y: i32); // error: could_not_infer

// Results whose type parameter is determined only by `out:` default to
// `NDArray<DTypeTag>` and therefore need `.asAnySpec` before re-entering a
// same-dtype operation:
final t_neg_add_multi_dot = add(multi_dot([f64, f64]), f64); // error: argument_type_not_assignable
final t_neg_matmul_schur = matmul(schur(f64).z, schur(f64).t); // error: argument_type_not_assignable

// Generic helpers: `T extends DTypeTag` / `T extends IntegerDType` do not
// satisfy the SelfOf bounds even for a single array used twice.
NDArray<T> badGenericAdd<T extends DTypeTag>(NDArray<T> a) => add(a, a); // error: argument_type_not_assignable
NDArray<T> badGenericGcd<T extends IntegerDType>(NDArray<T> a) => gcd(a, a); // error: argument_type_not_assignable

// Known limitation: `SelfOf<X>` does not imply `X`, so a `SelfOf`-bounded
// type parameter cannot be passed to the unary projection-bounded operations,
// and the capability markers cannot be passed to unary bitwise operations
// (`invert` / `~` require `RealOf<BitwiseDType>`, which the non-generic
// markers cannot implement because `RealOf` differs per integer tag).
NDArray<T> limitInvertSelfOf<T extends SelfOf<RealOf<IntegerDType>>>(NDArray<T> a) => invert(a); // error: argument_type_not_assignable
NDArray<R> limitSinSelfOf<T extends SelfOf<InexactOf<R>>, R extends DTypeTag>(NDArray<T> a) => sin(a); // error: argument_type_not_assignable
final t_neg_invert_lub = invert([i32, u8].first); // error: argument_type_not_assignable
final t_neg_opinv_lub = ~[i32, u8].first; // error: undefined_operator

// Negative subtype probes on capability markers:
void checkInvalidMarkerSubtypes() {
  isSubtype<Float64, BitwiseDType>(); // error: type_argument_not_matching_bounds
  isSubtype<Boolean, IntegerDType>(); // error: type_argument_not_matching_bounds
}

// Conflicting marker + DTypeSpec row witness (IntegerDType pins ElementOf<int>,
// so a row declaring Element = double fails with conflicting_generic_interfaces):
abstract final class _BadInt implements IntegerDType, DTypeSpec<_BadInt, double, Float64, Complex128, Float64, Int64, Float64, Float64> {} // error: conflicting_generic_interfaces
