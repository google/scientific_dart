// Generic helpers: `T extends DTypeTag` / `T extends IntegerDType` do not
// satisfy the SelfOf bounds even for a single array used twice.
NDArray<T> badGenericAdd<T extends DTypeTag>(NDArray<T> a) => add(a, a); // error: argument_type_not_assignable
NDArray<T> badGenericGcd<T extends IntegerDType>(NDArray<T> a) => gcd(a, a); // error: argument_type_not_assignable

// `SelfOf<X>` does not imply `X`: a `SelfOf`-bounded type parameter cannot be
// passed to the unary projection-bounded operations (bound by the row
// typedefs `InexactSpec<R>` / `AnyIntegerSpec` instead, see the positive
// fixture), and the capability markers cannot be passed to the bitwise
// *functions* (`invert` requires `RealOf<BitwiseDType>`, which the markers
// cannot implement because `RealOf` differs per integer tag) — only to the
// operators.
NDArray<T> limitInvertSelfOf<T extends SelfOf<RealOf<IntegerDType>>>(NDArray<T> a) => invert(a); // error: argument_type_not_assignable
NDArray<R> limitSinSelfOf<T extends SelfOf<InexactOf<R>>, R extends DTypeTag>(NDArray<T> a) => sin(a); // error: argument_type_not_assignable
final t_neg_invert_lub = invert([i32, u8].first); // error: argument_type_not_assignable
final t_neg_opand_anySpec = anySpec & anySpec; // error: undefined_operator

// Row-typedef bounds still reject the LUB of distinct tags:
NDArray<R> rowSinOfSum<T extends InexactSpec<R>, R extends DTypeTag>(NDArray<T> a, NDArray<T> b) {
  final s = add(a, b);
  return sin(s);
}

final t_neg_row_mixed_f = rowSinOfSum(f64, f32); // error: could_not_infer
